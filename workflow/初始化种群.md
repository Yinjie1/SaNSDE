    ```python
    class Population():
        '''种群对象
        ...
        Parameters
        ----------
        ...
        Examples
        --------
        >>> ...
        '''
        self._clusters: list[Atoms]
        @property
        def clusters(self):
            return self._clusters
        @property
        def symbol(self):
            return _symbol
        @symbol.setter
        def symbol(self):
            if symbol已经被设置:
                raise symbol已经被设置!
            self._symbol: str | None = 如二元合金团簇AnBm, 可用ase的symbols方法
        @property
        def size(self):
            return len(self._clusters)
        @property
        def num(self):
            return 根据化学式symbol推出一个团簇的原子数(int), 如AnBm则为n+m
        self.energies: list[float]
        self.forces: list[list[float]]
        def addCluster(self, cluster: Atoms, format: str = xyz):
            '''导入一个团簇对象(Atoms类型)到_clusters列表末尾
            Parameters
            ----------
            cluster: Atoms
            '''
            导入一个团簇对象(Atoms类型)到_clusters列表末尾, 必须确保种群中所有团簇化学式相同(symbol), 即每种元素的原子数分别相同. \
            团簇类型(symbol)可作为种群的一个属性, 可以在实例化种群时添加, 如果为空, 则在种群添加了第一个团簇时, 设为该团簇的化学 \
            式, `self.symbol = self.clusters[0].symbols`, 每次添加一个团簇都要判断化学式是否相同, 否则添加失败.
        def export(self, name: str = self):
            在当前目录'./Populations/'(如不存在则新建)下新建一个文件夹, 名称为name(缺省时为种群名称) \
            在文件夹中分别输出所有团簇结构文件(self.symbol+'_'+str(i).format), 包含化学式self.symbol, 能量和力self.energies[i], self.forces[i](如果有的话) \
            文件夹还包含一个说明文件(name.txt), 包含导出日期(YYYYMMddhhmmss), self.symbol, self.size, min(self.energies)和argmin(self.energies)(如果有的话)
        def import(self, name: str, confirm: boolean = False):
            读取'./Populations/name'种群文件, if 种群里没有团簇, 则直接覆盖, 如果至少有一个团簇, confirm = True时才覆盖, 否则加载失败!
    ```
