def crossover(pop1, pop2) -> Population:
    # 输入一个种群pop1, 和突变后的种群pop2, 输出交叉后的种群tempClusters
    tempClusters = Population(symbol = pop1.symbol)
    # tempClusters._clusters = pop1.clusters
    tempClusters.fp = pop1.fp
    tempClusters.p = pop1.p
    tempClusters.CRm = pop1.CRm
    # 第i个团簇交叉率
    CR[i]=random.gauss(CRm, 0.1)
    # if 第i个团簇进入下一代, then CRrec.apend(CR[i]), CRm=np.average(CRrec, weights=...)
    for i in range(pop1.size):
        # 随机选择的一个原子索引，确保至少有一个原子来自变异团簇
        jrand = np.random.randint(0,pop1.num-1)
        for j in range(pop1.num):
            r = np.random.random()
            cluster = Atoms()
            if r < CR[i] or j = jrand:
                cluster.append(pop2.clusters[i,j])
            else:
                cluster.append(pop1.clusters[i,j])
        tempClusters.addcluster(cluster)
    return tempClusters
