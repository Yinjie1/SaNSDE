```python
        for cluster in self.clusters:

    # 种群也是可索引的
    def __getitem__(self, i: int):
        return self.clusters[i]
    # 将逐步弃用原来的self.size
    def __len__(self):
        return len(self._clusters)
    ...
    # 能量和力将从团簇对象的属性中提取, 如不存在设为None, 以防搞乱索引顺序
    @property
    def energies(self) -> list[float | None]:
        _energies = []
        for cluster in self:
            if cluster.energy not None:
                _energies.append(cluster.energy)
            else:
                _energies.append(None)
        return self._energies
    @property
    def forces(self) -> list[list[list[float]] | None]:
        _forces = []
        for cluster in self:
            if cluster.energy not None:
                _forces.append(cluster.force)
            else:
                _forces.append(None)
        return self._forces
    ...
```

## 变异
```python
def mutate(pop: Population) -> Population:
    tempClusters = copy.deepcopy(pop)
    # 作用到整个种群pop上, 返回突变后的暂定种群tempClusters
    for i in range(len(pop)):
        # 产生随机数0~pop.num-1作为参与变异策略团簇的索引值, 并确保三个索引值互异: index1，index2，index3
        index1，index2，index3 = np.random.choice(pop.num, size=3, replace=False)
        # 确定第i个团簇变异因子F[i], 生成[0,1]随机数, for i in range(pop.size): if<fp: F[i]=random.gauss(0.5, 0.3), else: F[i]=np.random.standard_cauchy()
        if np.random.uniform(0,1) < tempClusters.fp:
            F = random.gauss(0.5, 0.3)
        else:
            F = np.random.standard_cauchy()
        # 确定变异策略进行变异操作, 生成[0,1]随机数, for i in range(pop.size): if<p, tempClusters.addcluster(DE/rand/1), else: DE/current-to-best/2
        if np.random.uniform(0,1) < p:
            tempClusters[i].positions = pop[index1].getpositions()+F*(pop[index2].getpositions()-pop[index3].getpositions())
            # 记录用什么方法突变, 0:'DE/rand/1', 1:'DE/current-to-best/2'
            tempClusters[i].mut = 0
        else:
            # 获取最低能量团簇索引
            indexBest = pop.index(min(pop.energies))
            tempClusters[i].positions = tempClusters[i].getpositions()+F*(pop[indexBest].getpositions()-tempClusters[i].getpositions())+F*(pop[index1].getpositions()-pop[index2].getpositions())
            tempClusters[i].mut = 1
    return tempClusters
```

## 交叉
```python
def crossover(pop1: Population, pop2: Population) -> Population:
    # 输入一个种群pop1, 和突变后的种群pop2, 输出交叉后的种群tempClusters
    # 需检查两个种群化学式, 团簇个数是否相同
    tempClusters = Population(symbol = pop1.symbol)
    tempClusters.fp = pop1.fp
    tempClusters.p = pop1.p
    tempClusters.CRm = pop1.CRm
    CR = np.zero(len(pop1))
    # if 第i个团簇进入下一代, then CRrec.apend(CR[i]), CRm=np.average(CRrec, weights=...)
    for i in range(len(pop1)):
        # 随机选择的一个原子索引，确保至少有一个原子来自变异团簇
        jrand = np.random.randint(0,pop1.num-1)
        # 第i个团簇交叉率
        CR[i]=np.random.gauss(CRm, 0.1)
        for j in range(pop1.num):
            r = np.random.random()
            cluster = Atoms()
            # Indexing an Atoms object returns an Atom object still remembering that it belongs to the collective Atoms: Modifying it will also change the Atoms object
            if r < CR[i] or j = jrand:
                # 请确保不是指向和原团簇相同的原子对象, 有必要的话就用深拷贝
                cluster.append(copy.deepcopy(pop2[i][j]))
            else:
                cluster.append(copy.deepcopy(pop1[i][j]))
        tempClusters.addcluster(cluster)
    tempClusters.CR = CR
    return tempClusters
```
